17ColoringResult SpillingAlgorithm::run(const Graph<int> &graph,
18 const std::vector<Web> &webs,
19 int numRegisters, int maxSpills) {
20 ColoringResult result;
21 result.success = true;
22 result.numColorsUsed = 0;
23
24 // ------------------------------------------------------------------
25 // Build auxiliary structures used throughout the allocation process.
26 // ------------------------------------------------------------------
27
28 std::unordered_map<int, const Web *>
29 webMap; // Maps each web ID to its corresponding Web object.
30 for (const auto &w : webs) {
31 webMap[w.id] = &w;
32 result.webToRegister[w.id] = -1;
33 }
34
35 std::unordered_set<int>
36 active; // Tracks all currently active nodes in the interference graph.
37 for (const auto &web : webs)
38 active.insert(web.id);
39
40 // ------------------------------------------------------------------
41 // Phase 0 - Initialization: O(V + E)
42 // ------------------------------------------------------------------
43 // Compute the initial degree of every active node.
44 // Nodes with degree strictly less than K (number of registers) are considered
45 // trivially colorable and are immediately added to the simplification list.
46 // ------------------------------------------------------------------
47 std::unordered_map<int, int> currentDegrees;
48 std::vector<int> trivialNodes;
49
50 for (int id : active) {
51 int deg = 0;
52 auto *v = graph.findVertex(id);
53 if (v) {
54 for (auto *edge : v->getAdj()) {
55 if (active.count(edge->getDest()->getInfo())) {
56 deg++;
57 }
58 }
59 }
60 currentDegrees[id] = deg;
61
62 // Nodes with degree < K are guaranteed to be colorable.
63 if (deg < numRegisters) {
64 trivialNodes.push_back(id);
65 }
66 }
67
68 std::stack<int> colorStack;
69 std::vector<int> forcedSpills;
70
71 // ------------------------------------------------------------------
72 // Phase 1 - Simplification and Spill Selection: O(V + E)
73 // ------------------------------------------------------------------
74 // Repeatedly:
75 // 1. Remove trivial nodes.
76 // 2. If none exist, select a spill candidate.
77 //
78 // Removed nodes are pushed onto a stack so they can later be
79 // reinserted in reverse order during coloring.
80 // ------------------------------------------------------------------
81
82 while (!active.empty()) {
83
84 // --------------------------------------------------------------
85 // Case 1: A trivially colorable node exists.
86 // --------------------------------------------------------------
87 if (!trivialNodes.empty()) {
88 // Extract a trivial node from the list
89 int id = trivialNodes.back();
90 trivialNodes.pop_back();
91
92 // It might have already been removed due to an earlier Spill
93 if (!active.count(id))
94 continue;
95
96 // Push the node onto the color stack, maintaining the simplification
97 // order
98 colorStack.push(id);
99 active.erase(id);
100
101 // Decrement the degree of all its active neighbors
102 auto *v = graph.findVertex(id);
103 if (v) {
104 for (auto *edge : v->getAdj()) {
105 int neighborId = edge->getDest()->getInfo();
106 if (active.count(neighborId)) {
107 currentDegrees[neighborId]--;
108 // If a neighbor's degree just dropped below K, it has become
109 // trivial. Push it to the list.
110 if (currentDegrees[neighborId] == numRegisters - 1) {
111 trivialNodes.push_back(neighborId);
112 }
113 }
114 }
115 }
116 } else {
117 // --------------------------------------------------------------
118 // Case 2: No trivial nodes exist.
119 // A spill is required to continue simplification.
120 // --------------------------------------------------------------
121
122 // Prevent excessive spilling.
123 if ((int)forcedSpills.size() >= maxSpills) {
124 std::cerr << "WARNING: Spill limit of " << maxSpills
125 << " exceeded. Allocation not possible.\n";
126 result.success = false;
127 return result;
128 }
129
130 // Select the best spill candidate using Chaitin's heuristic.
131 int candidate = selectSpillCandidate(active, webMap, currentDegrees);
132
133 if (candidate == -1 || !webMap.count(candidate)) {
134 result.success = false;
135 return result;
136 }
137
138 forcedSpills.push_back(candidate);
139 active.erase(candidate);
140 result.webToRegister[candidate] = -1;
141
142 std::cout << " Spilling web " << candidate << " ["
143 << webMap.at(candidate)->originalVariable << "] to memory.\n";
144
145 // ----------------------------------------------------------
146 // Removing a spilled node also decreases the degree of all
147 // its active neighbors.
148 // ----------------------------------------------------------
149 auto *v = graph.findVertex(candidate);
150 if (v) {
151 for (auto *edge : v->getAdj()) {
152 int neighborId = edge->getDest()->getInfo();
153 if (active.count(neighborId)) {
154 currentDegrees[neighborId]--;
155 // If the neighbor of the spilled node just became trivial, add it
156 // to the list.
157 if (currentDegrees[neighborId] == numRegisters - 1) {
158 trivialNodes.push_back(neighborId);
159 }
160 }
161 }
162 }
163 }
164 }
165
166 // ------------------------------------------------------------------
167 // Phase 2 - Graph Coloring
168 // ------------------------------------------------------------------
169 // Nodes are reinserted in reverse order of removal.
170 // Each node receives the first available register not already
171 // used by one of its adjacent neighbors.
172 // ------------------------------------------------------------------
173 std::unordered_map<int, int> colorAssignment;
174
175 while (!colorStack.empty()) {
176
177 // Retrieve the next node to color.
178 int id = colorStack.top();
179 colorStack.pop();
180
181 auto *v = graph.findVertex(id);
182
183 // Tracks which colors are already used by adjacent nodes.
184 std::unordered_set<int> usedColors;
185
186 if (v) {
187 for (auto *edge : v->getAdj()) {
188 int neighborId = edge->getDest()->getInfo();
189
190 // Check whether the neighbor has already been assigned a color.
191 auto it = colorAssignment.find(neighborId);
192 if (it != colorAssignment.end() && it->second >= 0)
193 usedColors.insert(it->second);
194 }
195 }
196
197 // --------------------------------------------------------------
198 // Select the first available register.
199 // --------------------------------------------------------------
200 int color = -1;
201 for (int c = 0; c < numRegisters; c++) {
202 if (!usedColors.count(c)) {
203 color = c;
204 break;
205 }
206 }
207
208 if (color == -1) {
209 // This state is theoretically unreachable if Phase 1 was correctly
210 // executed as nodes are only placed onto the stack if they are guaranteed
211 // to find a color.
212 std::cerr << "ERROR: Could not assign color to web " << id << "\n";
213 result.success = false;
214 return result;
215 }
216
217 // Assign the selected register to the current node.
218 colorAssignment[id] = color;
219
220 // Track the highest register index used.
221 result.numColorsUsed = std::max(result.numColorsUsed, color + 1);
222 }
223
224 // ------------------------------------------------------------------
225 // Store final register assignments.
226 // ------------------------------------------------------------------
227 for (const auto &[id, color] : colorAssignment)
228 result.webToRegister[id] = color;
229
230 result.spilledWebs = forcedSpills;
231 return result;
232}

234// ------------------------------------------------------------------
235// Calculates the spill cost of a web.
236//
237// Spill cost is estimated using the total number of definitions
238// and uses associated with the web.
239//
240// Frequently used webs are more expensive to spill because they
241// generate more memory load/store operations.
242// ------------------------------------------------------------------
243
244double SpillingAlgorithm::calculateSpillCost(const Web &web) {
245 return static_cast<double>(web.defLines.size() + web.useLines.size());
246}

248// ------------------------------------------------------------------
249// Selects the best spill candidate using Chaitin's heuristic:
250//
251// Spill Cost
252// Metric = -----------------
253// Active Degree
254//
255// - Spill Cost:
256// Estimated penalty of moving the web to memory.
257//
258// - Active Degree:
259// Number of current interferences in the active graph.
260//
261// Intuitively, we prefer spilling nodes with:
262// - low spill cost;
263// - high degree.
264//
265// This minimizes memory overhead while removing many
266// interference constraints from the graph.
267// ------------------------------------------------------------------
268
269int SpillingAlgorithm::selectSpillCandidate(
270 const std::unordered_set<int> &active,
271 const std::unordered_map<int, const Web *> &webMap,
272 const std::unordered_map<int, int> &currentDegrees) {
273
274 int bestCandidate = -1;
275 double minMetric = std::numeric_limits<double>::max();
276
277 // Iterate through all active nodes and select the one
278 // with the lowest spill heuristic value (cost / degree).
279 for (int id : active) {
280 auto it = webMap.find(id);
281 if (it == webMap.end())
282 continue;
283
284 int degree = currentDegrees.at(id);
285 // Prevent division by zero for isolated nodes.
286 // A degree of 0 is treated as 1 so that the
287 // metric depends only on spill cost.
288 if (degree == 0)
289 degree = 1;
290
291 double cost = calculateSpillCost(*it->second);
292 double metric = cost / degree;
293
294 // Greedily keep the node with the minimum metric.
295 if (metric < minMetric) {
296 minMetric = metric;
297 bestCandidate = id;
298 }
299 }
300
301 return bestCandidate;
302}
