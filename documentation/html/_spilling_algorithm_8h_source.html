1/**
2 * @file SpillingAlgorithm.h
3 * @brief Declares SpillingAlgorithm — a Chaitin-style register allocator
4 * that explicitly spills webs to memory when necessary.
5 *
6 * Algorithm overview:
7 * - **Phase 0 (Init)**: compute active degrees; identify trivially colorable nodes.
8 * - **Phase 1 (Simplification)**: remove trivial nodes (degree < K) onto a stack;
9 * when none exist, select and remove a spill candidate using Chaitin's
10 * min(cost/degree) heuristic.
11 * - **Phase 2 (Coloring)**: pop nodes from the stack and assign the lowest
12 * free register color.
13 *
14 */

31 /**
32 * @brief Run register allocation with spilling.
33 *
34 * Executes the three-phase Chaitin-style allocator. The process terminates
35 * early (returning `result.success = false`) if more than `maxSpills` webs
36 * would need to be spilled.
37 *
38 * @param graph Interference graph (vertices keyed by web ID).
39 * @param webs All webs to allocate.
40 * @param numRegisters Number of available physical registers (K colors).
41 * @param maxSpills Maximum number of webs that may be spilled.
42 * Allocation fails immediately if this limit is exceeded.
43 * @return ColoringResult with `success`, `numColorsUsed`, `webToRegister`
44 * (register index or -1 for spilled webs), and `spilledWebs`.
45 *
46 * @complexity O(V + E) where V = number of webs and E = number of
47 * interference edges (each node and edge processed O(1) times).
48 */
49 static ColoringResult run(const Graph<int> &graph,
50 const std::vector<Web> &webs, int numRegisters,
51 int maxSpills);
52
53private:
54 /**
55 * @brief Calculate the spill cost for a web.
56 *
57 * Cost is defined as `(|defLines| + |useLines|)`. Webs that are defined
58 * and used frequently are costly to spill because each spill generates
59 * additional load/store memory operations.
60 *
61 * @param web The web whose spill cost is computed.
62 * @return Spill cost as a double (always ≥ 0).
63 *
64 * @complexity O(1)
65 */
66 static double calculateSpillCost(const Web &web);
67
68 /**
69 * @brief Select the best spill candidate using Chaitin's heuristic.
70 *
71 * Evaluates all currently active nodes and returns the one that minimises:
72 * @code
73 * Spill Cost
74 * metric = ---------------
75 * Active Degree
76 * @endcode
77 * A node with a low metric is cheap to remove (low cost, many constraints
78 * resolved). Isolated nodes (degree 0) use an effective degree of 1 to
79 * avoid division by zero.
80 *
81 * @param active Set of currently active web IDs.
82 * @param webMap Map from web ID to its Web metadata pointer.
83 * @param currentDegrees Map from web ID to its current active-graph degree.
84 * @return ID of the best spill candidate, or -1 if @p active is empty.
85 *
86 * @complexity O(V) where V = |active|.
87 */
88 static int selectSpillCandidate(const std::unordered_set<int> &active,
89 const std::unordered_map<int, const Web *> &webMap,
90 const std::unordered_map<int, int> &currentDegrees);
91};
