16/**
17 * @brief Trims leading and trailing whitespace from a string.
18 *
19 * Used internally by loadRegisters() and loadRanges() to sanitise
20 * tokens read from input files.
21 *
22 * @param str Input string.
23 * @return A copy of @p str with leading/trailing whitespace removed.
24 *
25 * @complexity O(N) where N is the length of the string.
26 */
27std::string trim(const std::string &str) {
28 auto start = std::find_if_not(
29 str.begin(), str.end(), [](unsigned char c) { return std::isspace(c); });
30 auto end = std::find_if_not(str.rbegin(), str.rend(), [](unsigned char c) {
31 return std::isspace(c);
32 }).base();
33 return (start < end) ? std::string(start, end) : std::string();
34}

37std::string DataLoader::resolvePath(const std::string &inputPath,

39 if (inputPath.find('/') != std::string::npos ||
40 inputPath.find('\\') != std::string::npos) {

61RegisterConfig DataLoader::loadRegisters(const std::string &filename) {
62 std::ifstream file(filename);
63 if (!file.is_open()) {
64 std::cerr << "Error: Could not open file " << filename << std::endl;
65 throw std::runtime_error("Could not open file " + filename);
66 }
67
68 RegisterConfig config;
69 std::string line;
70 int linesParsed = 0;
71
72 while (std::getline(file, line)) {
73 line = trim(line);
74 // Ignore empty lines and comments
75 if (line.empty() || line[0] == '#') {
76 continue;
77 }
78
79 std::istringstream iss(line);
80 std::string key;
81 if (std::getline(iss, key, ':')) {
82 key = trim(key);
83 std::string valueStr;
84 if (std::getline(iss, valueStr)) {
85 valueStr = trim(valueStr);
86
87 if (key == "registers") {
88 try {
89 config.numRegisters = std::stoi(valueStr);
90 linesParsed++;
91 } catch (const std::exception &e) {
92 std::cerr << "Error: Bad format for registers in " << filename
93 << std::endl;
94 throw std::runtime_error("Bad format in " + filename);
95 }
96 } else if (key == "algorithm") {
97 size_t commaPos = valueStr.find(',');
98
99 if (commaPos != std::string::npos) {
100 config.algorithmName = trim(valueStr.substr(0, commaPos));
101 try {
102 config.algorithmParam =
103 std::stoi(trim(valueStr.substr(commaPos + 1)));
104 } catch (const std::exception &e) {
105 std::cerr << "Error: Bad format for algorithm parameter in "
106 << filename << std::endl;
107 throw std::runtime_error("Bad format in " + filename);
108 }
109 } else {
110 config.algorithmName = valueStr;
111
112 if (config.algorithmName == "spilling" ||
113 config.algorithmName == "splitting") {
114
115 throw std::runtime_error(config.algorithmName +
116 " requires a parameter K (e.g. " +
117 config.algorithmName + ", 2)");
118 }
119 }
120 linesParsed++;
121 }
122 }
123 } else {
124 std::cerr << "Error: Malformed line in " << filename << ": " << line
125 << std::endl;
126 throw std::runtime_error("Bad format in " + filename);
127 }
128 }
129
130 if (linesParsed == 0) {
131 std::cerr << "Error: File " << filename
132 << " appears to be empty or malformed." << std::endl;
133 throw std::runtime_error("Bad format in " + filename);
134 }
135
136 return config;
137}

139std::unordered_map<std::string, VariableRanges>
140DataLoader::loadRanges(const std::string &filename) {
141 std::ifstream file(filename);
142 if (!file.is_open()) {
143 std::cerr << "Error: Could not open file " << filename << std::endl;
144 throw std::runtime_error("Could not open file " + filename);
145 }
146
147 std::unordered_map<std::string, VariableRanges> rangesMap;
148 std::string line;
149
150 while (std::getline(file, line)) {
151 line = trim(line);
152 // Ignore empty lines and comments
153 if (line.empty() || line[0] == '#') {
154 continue;
155 }
156
157 std::istringstream iss(line);
158 std::string varName;
159 if (std::getline(iss, varName, ':')) {
160 varName = trim(varName);
161 if (varName.empty()) {
162 std::cerr << "Error: Missing variable name in " << filename
163 << std::endl;
164 throw std::runtime_error("Malformed ranges file format");
165 }
166
167 std::string rangesStr;
168 if (!std::getline(iss, rangesStr)) {
169 std::cerr << "Error: Missing ranges for variable '" << varName
170 << "' in " << filename << std::endl;
171 throw std::runtime_error("Malformed ranges file format");
172 }
173
174 std::vector<LiveRangePoint> currentRanges;
175 std::istringstream rss(rangesStr);
176 std::string token;
177
178 while (std::getline(rss, token, ',')) {
179 token = trim(token);
180 if (token.empty())
181 continue;
182
183 LiveRangePoint point = {0, false, false};
184
185 if (token.back() == '+') {
186 point.isDef = true;
187 token.pop_back();
188 } else if (token.back() == '-') {
189 point.isEnd = true;
190 token.pop_back();
191 }
192
193 try {
194 point.line = std::stoi(token);
195 currentRanges.push_back(point);
196 } catch (const std::exception &e) {
197 std::cerr << "Error: Malformed line number '" << token
198 << "' for variable '" << varName << "' in " << filename
199 << std::endl;
200 throw std::runtime_error("Malformed ranges file format");
201 }
202 }
203
204 if (!currentRanges.empty()) {
205 auto &varRanges = rangesMap[varName];
206 if (varRanges.varName.empty()) {
207 varRanges.varName = varName;
208 }
209 varRanges.ranges.push_back(currentRanges);
210 }
211 } else {
212 std::cerr << "Error: Malformed line in " << filename << ": " << line
213 << std::endl;
214 throw std::runtime_error("Malformed ranges file format");
215 }
216 }
217
218 return rangesMap;
219}
