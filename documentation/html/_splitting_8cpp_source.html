18/**
19 * @brief Determines whether two webs interfere with each other.
20 *
21 * Two webs interfere if they share at least one active line where the
22 * combination does not satisfy the exception rule (one web uses the line
23 * as its last-use point while the other defines it at the same line).
24 *
25 * @param a First web.
26 * @param b Second web.
27 * @return `true` if the webs interfere and cannot share a register;
28 * `false` if their live ranges are completely disjoint or the
29 * overlap is entirely at adjacent use/def boundaries.
30 *
31 * @complexity O(L_a + L_b) amortised due to sorted set_intersection
32 * followed by O(S) binary searches (S = |shared lines|).
33 */
34bool interferes(const Web& a, const Web& b) {
35 std::vector<int> sharedLines;
36 std::set_intersection(a.activeLines.begin(), a.activeLines.end(),
37 b.activeLines.begin(), b.activeLines.end(),
38 std::back_inserter(sharedLines));
39
40 // No common lines -> no interference
41 if (sharedLines.empty()) {
42 return false;
43 }
44
45 bool allExceptions = true;
46 for (int L : sharedLines) {
47 bool aUsesBDefs = std::binary_search(a.useLines.begin(), a.useLines.end(), L) &&
48 std::binary_search(b.defLines.begin(), b.defLines.end(), L);
49
50 bool bUsesADefs = std::binary_search(b.useLines.begin(), b.useLines.end(), L) &&
51 std::binary_search(a.defLines.begin(), a.defLines.end(), L);
52
53 if (!aUsesBDefs && !bUsesADefs) {
54 allExceptions = false;
55 break;
56 }
57 }
58
59 return !allExceptions;
60 }
61}

63ColoringResult Splitting::runWithSplitting(
64 const Graph<int>& graph,
65 const std::vector<Web>& webs,
66 int numRegisters,
67 int maxSplits)
68{
69 // ------------------------------------------------------------------
70 // Iteratively split webs and retry coloring.
71 // Each round: O(V+E) candidate selection + O(L*W) split point
72 // + O(V^2) graph rebuild + O(V^2) coloring
73 // Total: O(maxSplits * (V^2 + L*W))
74 // ------------------------------------------------------------------
75 ColoringResult result;
76 std::vector<Web> currentWebs = webs;
77 Graph<int> currentGraph = GraphBuilder::buildInterferenceGraph(currentWebs);
78
79 // Compute next available web ID (continue from highest existing + 1)
80 int nextWebId = 0;
81 for (const auto& w : currentWebs) {
82 nextWebId = std::max(nextWebId, w.id + 1);
83 }
84
85 for (int round = 1; round <= maxSplits; ++round) {
86
87 // (a) Select the web with the highest degree
88 int candidateId = selectSplitCandidate(currentGraph, currentWebs);
89 if (candidateId == -1) {
90 std::cerr << " Splitting: No candidate found at round "
91 << round << ".\n";
92 break;
93 }
94
95 // Find the Web object for the candidate
96 const Web* candidateWeb = nullptr;
97 size_t candidateIdx = 0;
98 for (size_t i = 0; i < currentWebs.size(); ++i) {
99 if (currentWebs[i].id == candidateId) {
100 candidateWeb = &currentWebs[i];
101 candidateIdx = i;
102 break;
103 }
104 }
105 if (!candidateWeb) {
106 std::cerr << " Splitting: Candidate web " << candidateId
107 << " not found in web list.\n";
108 break;
109 }
110
111 // (b) Find the optimal split point
112 int cutIdx = findSplitPoint(*candidateWeb, currentWebs);
113 if (cutIdx <= 0) {
114 // Cannot split this web (too few active lines)
115 std::cout << " Splitting: Web " << candidateId
116 << " cannot be split (too few active lines). Skipping.\n";
117 continue;
118 }
119
120 // (c) Create two sub-webs from the split
121 Web webA;
122 webA.id = nextWebId++;
123 webA.originalVariable = candidateWeb->originalVariable;
124
125 Web webB;
126 webB.id = nextWebId++;
127 webB.originalVariable = candidateWeb->originalVariable;
128
129 // web_a gets activeLines [0..cutIdx-1], web_b gets [cutIdx..end]
130 webA.activeLines.assign(
131 candidateWeb->activeLines.begin(),
132 candidateWeb->activeLines.begin() + cutIdx);
133
134 webB.activeLines.assign(
135 candidateWeb->activeLines.begin() + cutIdx,
136 candidateWeb->activeLines.end());
137
138 // Inherit defLines and useLines only for the lines each sub-web contains
139 std::unordered_set<int> linesA(webA.activeLines.begin(),
140 webA.activeLines.end());
141 std::unordered_set<int> linesB(webB.activeLines.begin(),
142 webB.activeLines.end());
143
144 for (int line : candidateWeb->defLines) {
145 if (linesA.count(line)) webA.defLines.push_back(line);
146 if (linesB.count(line)) webB.defLines.push_back(line);
147 }
148 for (int line : candidateWeb->useLines) {
149 if (linesA.count(line)) webA.useLines.push_back(line);
150 if (linesB.count(line)) webB.useLines.push_back(line);
151 }
152
153 // Ensure sorted order (activeLines already come sorted from the
154 // original web; defLines and useLines are filtered in order)
155 std::sort(webA.defLines.begin(), webA.defLines.end());
156 std::sort(webA.useLines.begin(), webA.useLines.end());
157 std::sort(webB.defLines.begin(), webB.defLines.end());
158 std::sort(webB.useLines.begin(), webB.useLines.end());
159
160 std::cout << " Splitting round " << round << ": web "
161 << candidateId << " ["
162 << candidateWeb->originalVariable << "] -> web "
163 << webA.id << " (lines "
164 << webA.activeLines.front() << "-"
165 << webA.activeLines.back() << ") + web "
166 << webB.id << " (lines "
167 << webB.activeLines.front() << "-"
168 << webB.activeLines.back() << ")\n";
169
170 // (d) Replace the original web with the two sub-webs
171 currentWebs.erase(currentWebs.begin() + candidateIdx);
172 currentWebs.push_back(webA);
173 currentWebs.push_back(webB);
174
175 // (e) Update interference graph incrementally instead of rebuilding from scratch
176 currentGraph.removeVertex(candidateId);
177 currentGraph.addVertex(webA.id);
178 currentGraph.addVertex(webB.id);
179
180 for (const auto& other : currentWebs) {
181 if (other.id == webA.id || other.id == webB.id) continue;
182
183 if (interferes(webA, other)) {
184 currentGraph.addBidirectionalEdge(webA.id, other.id, 1.0);
185 }
186 if (interferes(webB, other)) {
187 currentGraph.addBidirectionalEdge(webB.id, other.id, 1.0);
188 }
189 }
190 if (interferes(webA, webB)) {
191 currentGraph.addBidirectionalEdge(webA.id, webB.id, 1.0);
192 }
193
194 // (f) Try coloring again
195 result = GraphColoring::run(currentGraph, currentWebs, numRegisters, {});
196 if (result.success) {
197 std::cout << " Splitting: Coloring succeeded after "
198 << round << " split(s).\n";
199 result.splitWebs = currentWebs;
200 return result;
201 }
202 }
203
204 // ------------------------------------------------------------------
205 // Step 3: All splits exhausted — return failure with all webs in memory.
206 // ------------------------------------------------------------------
207 std::cerr << " Splitting: Coloring failed after " << maxSplits
208 << " split(s). Assigning all webs to memory.\n";
209
210 result.success = false;
211 result.numColorsUsed = 0;
212 result.webToRegister.clear();
213 result.spilledWebs.clear();
214
215 for (const auto& w : currentWebs) {
216 result.webToRegister[w.id] = -1;
217 result.spilledWebs.push_back(w.id);
218 }
219
220 result.splitWebs = currentWebs;
221 return result;
222}

224int Splitting::selectSplitCandidate(
225 const Graph<int>& graph,
226 const std::vector<Web>& webs)
227{
228 std::unordered_set<int> webIds;
229 for (const auto& w : webs) {
230 webIds.insert(w.id);
231 }
232
233 // Build a map from web ID to number of active lines for tie-breaking
234 std::unordered_map<int, int> webSize;
235 for (const auto& w : webs) {
236 webSize[w.id] = static_cast<int>(w.activeLines.size());
237 }
238
239 int bestId = -1;
240 int bestDegree = -1;
241 int bestSize = -1;
242
243 for (auto* v : graph.getVertexSet()) {
244 int id = v->getInfo();
245 if (!webIds.count(id)) continue;
246
247 int deg = 0;
248 for (auto* edge : v->getAdj()) {
249 if (webIds.count(edge->getDest()->getInfo())) {
250 deg++;
251 }
252 }
253
254 int size = webSize.count(id) ? webSize[id] : 0;
255
256 // Prefer higher degree; break ties by preferring longer webs
257 if (deg > bestDegree || (deg == bestDegree && size > bestSize)) {
258 bestDegree = deg;
259 bestId = id;
260 bestSize = size;
261 }
262 }
263
264 return bestId;
265}

267int Splitting::findSplitPoint(
268 const Web& web,
269 const std::vector<Web>& allWebs)
270{
271 const auto& lines = web.activeLines;
272 int n = static_cast<int>(lines.size());
273
274 // Need at least 2 active lines to split
275 if (n < 2) return -1;
276
277 // For each candidate cut position (1..n-1), the split point is
278 // "before lines[cutIdx]". We pick the cut that minimises the
279 // number of other webs active at that line (interference pressure).
280 //
281 // Time Complexity: O(n * W) where W = |allWebs|
282 int bestCutIdx = 1; // default: after the first line
283 int minInterference = std::numeric_limits<int>::max();
284 int midpoint = n / 2;
285
286 for (int cutIdx = 1; cutIdx < n; ++cutIdx) {
287 int line = lines[cutIdx];
288 int interference = 0;
289
290 // Count how many OTHER webs are active at this line
291 for (const auto& other : allWebs) {
292 if (other.id == web.id) continue;
293
294 // other.activeLines is sorted, so binary search
295 if (std::binary_search(other.activeLines.begin(),
296 other.activeLines.end(),
297 line)) {
298 interference++;
299 }
300 }
301
302 // Prefer lower interference; break ties by preferring the
303 // cut closest to the midpoint (most balanced split)
304 int curDist = cutIdx > midpoint ? cutIdx - midpoint : midpoint - cutIdx;
305 int bestDist = bestCutIdx > midpoint ? bestCutIdx - midpoint : midpoint - bestCutIdx;
306 if (interference < minInterference ||
307 (interference == minInterference && curDist < bestDist)) {
308 minInterference = interference;
309 bestCutIdx = cutIdx;
310 }
311 }
312
313 return bestCutIdx;
314}
