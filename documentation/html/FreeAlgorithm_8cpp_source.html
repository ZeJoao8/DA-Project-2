18ColoringResult FreeAlgorithm::run(
19 const Graph<int>& graph,
20 const std::vector<Web>& webs,
21 int numRegisters)
22{
23 ColoringResult result;
24 result.success = true;
25 result.numColorsUsed = 0;
26
27 // Build active set and vertex lookup map once
28 std::unordered_set<int> active;
29 std::unordered_map<int, Vertex<int>*> vertexMap;
30 for (auto* v : graph.getVertexSet()) {
31 vertexMap[v->getInfo()] = v;
32 }
33
34 for (const auto& web : webs) {
35 active.insert(web.id);
36 }
37
38 // Dynamic degrees and trivial nodes initialization: O(V + E)
39 std::unordered_map<int, int> currentDegrees;
40 std::vector<int> trivialNodes;
41
42 for (int id : active) {
43 int deg = 0;
44 auto it = vertexMap.find(id);
45 if (it != vertexMap.end() && it->second != nullptr) {
46 for (auto* edge : it->second->getAdj()) {
47 if (active.count(edge->getDest()->getInfo())) {
48 deg++;
49 }
50 }
51 }
52 currentDegrees[id] = deg;
53
54 if (deg < numRegisters) {
55 trivialNodes.push_back(id);
56 }
57 }
58
59 std::stack<int> colorStack;
60
61 // Simplification Phase: O(V + E)
62 while (!active.empty()) {
63 if (!trivialNodes.empty()) {
64 int id = trivialNodes.back();
65 trivialNodes.pop_back();
66
67 if (!active.count(id)) {
68 continue;
69 }
70
71 colorStack.push(id);
72 active.erase(id);
73
74 // Decrement active neighbor degrees
75 auto it = vertexMap.find(id);
76 if (it != vertexMap.end() && it->second != nullptr) {
77 for (auto* edge : it->second->getAdj()) {
78 int neighborId = edge->getDest()->getInfo();
79 if (active.count(neighborId)) {
80 currentDegrees[neighborId]--;
81 if (currentDegrees[neighborId] == numRegisters - 1) {
82 trivialNodes.push_back(neighborId);
83 }
84 }
85 }
86 }
87 } else {
88 // Optimistic coloring: push highest-degree node onto stack anyway
89 int bestCandidate = -1;
90 int bestDegree = -1;
91 for (int id : active) {
92 int deg = currentDegrees[id];
93 if (deg > bestDegree) {
94 bestDegree = deg;
95 bestCandidate = id;
96 }
97 }
98
99 if (bestCandidate == -1) {
100 result.success = false;
101 break;
102 }
103
104 colorStack.push(bestCandidate);
105 active.erase(bestCandidate);
106
107 // Decrement active neighbor degrees of the optimistically pushed node
108 auto it = vertexMap.find(bestCandidate);
109 if (it != vertexMap.end() && it->second != nullptr) {
110 for (auto* edge : it->second->getAdj()) {
111 int neighborId = edge->getDest()->getInfo();
112 if (active.count(neighborId)) {
113 currentDegrees[neighborId]--;
114 if (currentDegrees[neighborId] == numRegisters - 1) {
115 trivialNodes.push_back(neighborId);
116 }
117 }
118 }
119 }
120 }
121 }
122
123 // -----------------------------------------------------------------------
124 // Phase 2 – Coloring with Kempe Chain rescue
125 // Pop nodes from the stack and assign the lowest free color.
126 // If no free color exists, attempt a Kempe Chain swap before spilling.
127 // Use vertexMap for fast lookup of adjacent nodes.
128 // Time Complexity: O(V * N^2 * (V + E)) in the worst case
129 // -----------------------------------------------------------------------
130 std::unordered_map<int, int> colorAssignment;
131
132 while (!colorStack.empty()) {
133 int id = colorStack.top();
134 colorStack.pop();
135
136 // Collect colors of already-assigned neighbors
137 std::vector<int> neighborColors;
138 auto itLookup = vertexMap.find(id);
139 if (itLookup != vertexMap.end() && itLookup->second != nullptr) {
140 for (auto* edge : itLookup->second->getAdj()) {
141 int neighborId = edge->getDest()->getInfo();
142 auto it = colorAssignment.find(neighborId);
143 if (it != colorAssignment.end() && it->second >= 0) {
144 neighborColors.push_back(it->second);
145 }
146 }
147 }
148
149 // Try simple color assignment first
150 std::unordered_set<int> usedColors(neighborColors.begin(), neighborColors.end());
151 int color = -1;
152 for (int c = 0; c < numRegisters; c++) {
153 if (!usedColors.count(c)) {
154 color = c;
155 break;
156 }
157 }
158
159 if (color == -1) {
160 // All colors taken — attempt Kempe Chain rescue
161 std::cout << " [Kempe] Node " << id
162 << " stuck (all " << numRegisters
163 << " registers taken). Attempting rescue...\n";
164
165 color = tryKempeRescue(graph, id, colorAssignment, numRegisters);
166
167 if (color != -1) {
168 std::cout << " [Kempe] Rescue succeeded! Freed register "
169 << color << " for node " << id << ".\n";
170 } else {
171 std::cout << " [Kempe] Rescue failed. Spilling node " << id << ".\n";
172 result.success = false;
173 colorAssignment[id] = -1;
174 result.spilledWebs.push_back(id);
175 continue;
176 }
177 }
178
179 colorAssignment[id] = color;
180 result.numColorsUsed = std::max(result.numColorsUsed, color + 1);
181 }
182
183 // Merge colorAssignment into result
184 for (const auto& [id, color] : colorAssignment) {
185 result.webToRegister[id] = color;
186 }
187
188 return result;
189}

191// ---------------------------------------------------------------------------
192// buildKempeChain
193// BFS from startId, traversing only c1- or c2-colored nodes in the graph.
194// ---------------------------------------------------------------------------
195std::unordered_set<int> FreeAlgorithm::buildKempeChain(
196 const Graph<int>& graph,
197 const std::unordered_map<int, int>& colorAssignment,
198 int startId,
199 int c1,
200 int c2)
201{
202 std::unordered_set<int> chain;
203 std::queue<int> bfsQueue;
204
205 bfsQueue.push(startId);
206 chain.insert(startId);
207
208 while (!bfsQueue.empty()) {
209 int current = bfsQueue.front();
210 bfsQueue.pop();
211
212 auto* v = graph.findVertex(current);
213 if (!v) continue;
214
215 for (auto* edge : v->getAdj()) {
216 int neighborId = edge->getDest()->getInfo();
217 if (chain.count(neighborId)) continue; // already visited
218
219 auto it = colorAssignment.find(neighborId);
220 if (it == colorAssignment.end()) continue; // not yet colored
221
222 int neighborColor = it->second;
223 if (neighborColor == c1 || neighborColor == c2) {
224 chain.insert(neighborId);
225 bfsQueue.push(neighborId);
226 }
227 }
228 }
229
230 return chain;
231}

233// ---------------------------------------------------------------------------
234// tryKempeRescue
235// For each color c used by a neighbor of nodeId, try to swap it with every
236// other color c2 used by a different neighbor, validating that the chain
237// starting at the c-neighbor does NOT contain any c2-neighbor of nodeId.
238// ---------------------------------------------------------------------------
239int FreeAlgorithm::tryKempeRescue(
240 const Graph<int>& graph,
241 int nodeId,
242 std::unordered_map<int, int>& colorAssignment,
243 int numRegisters)
244{
245 auto* vNode = graph.findVertex(nodeId);
246 if (!vNode) return -1;
247
248 // Build neighbor -> color map (only already-colored neighbors)
249 // and group neighbors by color
250 std::unordered_map<int, std::vector<int>> colorToNeighbors; // color -> list of neighbor IDs
251 for (auto* edge : vNode->getAdj()) {
252 int neighborId = edge->getDest()->getInfo();
253 auto it = colorAssignment.find(neighborId);
254 if (it != colorAssignment.end() && it->second >= 0) {
255 colorToNeighbors[it->second].push_back(neighborId);
256 }
257 }
258
259 // Collect all colors that are actually saturating nodeId
260 std::vector<int> usedColorsList;
261 for (const auto& [color, _] : colorToNeighbors) {
262 usedColorsList.push_back(color);
263 }
264
265 // Try to free color c by swapping c <-> c2
266 for (int c : usedColorsList) {
267 const std::vector<int>& cNeighbors = colorToNeighbors.at(c);
268
269 for (int c2 : usedColorsList) {
270 if (c2 == c) continue;
271
272 const std::vector<int>& c2Neighbors = colorToNeighbors.at(c2);
273
274 // Try each c-colored neighbor of nodeId as the chain start
275 for (int nodeA : cNeighbors) {
276 std::unordered_set<int> chain =
277 buildKempeChain(graph, colorAssignment, nodeA, c, c2);
278
279 // Validate: no c2-neighbor of nodeId may be in the chain
280 bool valid = true;
281 for (int nodeB : c2Neighbors) {
282 if (chain.count(nodeB)) {
283 valid = false;
284 break;
285 }
286 }
287
288 if (valid) {
289 // Execute the swap: c <-> c2 for all nodes in the chain
290 for (int chainNode : chain) {
291 int& assignedColor = colorAssignment.at(chainNode);
292 assignedColor = (assignedColor == c) ? c2 : c;
293 }
294 // Color c is now free for nodeId
295 return c;
296 }
297 }
298 }
299 }
300
301 return -1; // No rescue possible
302}
