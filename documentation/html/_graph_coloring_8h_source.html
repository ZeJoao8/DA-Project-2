5 *
6 * Algorithm summary (Kempe / Chaitin simplification):
7 * - **Phase 1 (Simplification)**: iteratively remove nodes with active degree < K
8 * onto a stack. If all remaining nodes have degree ≥ K, spill the
9 * highest-degree node.
10 * - **Phase 2 (Coloring)**: pop the stack and greedily assign the lowest
11 * available color not yet used by any already-colored neighbor.

13 */

34/**
35 * @brief Greedy graph coloring algorithm for register allocation.
36 *
37 * Phase 1 - Simplification: iteratively remove nodes with degree < N onto a stack.
38 * If all remaining nodes have degree >= N, spill the highest-degree node.
39 * Phase 2 - Coloring: pop stack and assign the lowest available color not used by neighbors.
40 *
41 * @complexity O(V^2) where V = number of webs.
42 */
43class GraphColoring {
44public:
45 /**
46 * @brief Run the greedy coloring on the interference graph.
47 *
48 * @param graph Interference graph with web IDs as vertex info.
49 * @param webs All webs (for metadata).
50 * @param numRegisters Number of available registers (N colors).
51 * @param spilledWebIds Web IDs pre-decided as spilled (excluded from coloring).
52 * @return ColoringResult with success flag and web->register mapping.
53 *
54 * @complexity O(V^2)
55 */
56 static ColoringResult run(
57 const Graph<int>& graph,
58 const std::vector<Web>& webs,
59 int numRegisters,
60 const std::vector<int>& spilledWebIds = {}
61 );
62
63 /**
64 * @brief Select the best spill candidate: highest-degree node among active nodes.
65 *
66 * @param graph Interference graph.
67 * @param excludedIds Web IDs to skip (already spilled or on stack).
68 * @param activeIds Web IDs currently active in the working graph.
69 * @return Web ID of best candidate, or -1 if none found.
70 *
71 * @complexity O(V + E)
72 */
73 static int selectSpillCandidate(

87 static int assignColor(

89 int numColors
