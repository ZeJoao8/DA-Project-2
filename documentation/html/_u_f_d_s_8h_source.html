17/**
18 * @brief Union-Find Disjoint Set structure with path compression and
19 * union by rank.
20 *
21 * Supports near-constant-time set membership queries and merges.
22 * Uses path compression in findSet() and union-by-rank in linkSets()
23 * to achieve amortised O(α(N)) per operation (where α is the inverse
24 * Ackermann function — effectively O(1) in practice).
25 */
26class UFDS {
27public:
28 /**
29 * @brief Initialises N disjoint singleton sets {0}, {1}, ..., {N-1}.
30 *
31 * @param N Total number of elements.
32 * @complexity O(N)
33 */
34 UFDS(unsigned int N);
35
36 /**
37 * @brief Finds the representative (root) of the set containing element @p i.
38 *
39 * Applies **path compression**: after the call, all nodes on the path
40 * from @p i to the root point directly to the root.
41 *
42 * @param i Element index (0 ≤ i < N).
43 * @return The root element of the set containing @p i.
44 * @complexity O(α(N)) amortised.
45 */
46 unsigned long findSet(unsigned int i);
47
48 /**
49 * @brief Returns true if elements @p i and @p j belong to the same set.
50 *
51 * @param i First element index.
52 * @param j Second element index.
53 * @return `true` if findSet(i) == findSet(j).
54 * @complexity O(α(N)) amortised.
55 */
56 bool isSameSet(unsigned int i, unsigned int j);
57
58 /**
59 * @brief Merges the sets containing @p i and @p j (union by rank).
60 *
61 * If i and j are already in the same set, this is a no-op.
62 * The root with the **lower rank** becomes a child of the root with
63 * the higher rank. If ranks are equal, @p j's root becomes the new
64 * root and its rank is incremented.
65 *
66 * @param i First element index.
67 * @param j Second element index.
68 * @complexity O(α(N)) amortised.
69 */
70 void linkSets(unsigned int i, unsigned int j);
71
72private:
73 std::vector<unsigned int> path; ///< path[i] = parent of node i (itself if root).
74 std::vector<unsigned int> rank; ///< rank[i] = upper bound on the height of the subtree rooted at i.
75};
