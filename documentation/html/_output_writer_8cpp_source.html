15bool OutputWriter::writeResult(const std::string &filePath,
16 const std::vector<Web> &webs,
17 const ColoringResult &result,
18 int numAvailableRegisters) {
19 std::ofstream outFile(filePath);
20 if (!outFile.is_open()) {
21 return false;
22 }
23
24 outFile << "# Total number of webs followed by the listing of the program "
25 "points of each one\n";
26 outFile << "# program points in each web are sorted in ascending order\n";
27 outFile << "webs: " << webs.size() << "\n";
28
29 for (const auto &web : webs) {
30 outFile << "web" << web.id << ": ";
31
32 // Copy and sort active lines just in case
33 std::vector<int> sortedLines = web.activeLines;
34 std::sort(sortedLines.begin(), sortedLines.end());
35
36 for (size_t i = 0; i < sortedLines.size(); ++i) {
37 int line = sortedLines[i];
38 outFile << line;
39
40 // Check if it's a definition point
41 bool isDef = std::find(web.defLines.begin(), web.defLines.end(), line) !=
42 web.defLines.end();
43 // Check if it's an end point (use point that terminates the range)
44 bool isEnd = std::find(web.useLines.begin(), web.useLines.end(), line) !=
45 web.useLines.end();
46
47 if (isDef)
48 outFile << "+";
49 if (isEnd)
50 outFile << "-";
51
52 if (i < sortedLines.size() - 1) {
53 outFile << ",";
54 }
55 }
56 outFile << "\n";
57 }
58
59 outFile
60 << "# Total number of registers used, followed by assignment to webs\n";
61
62 if (!result.success) {
63 outFile << "registers: 0\n";
64 for (const auto &web : webs) {
65 outFile << "M: web" << web.id << "\n";
66 }
67 } else {
68 outFile << "registers: " << result.numColorsUsed << "\n";
69
70 // Output rX: webY for each allocated web
71 // Note: The example shows r0: web0, r0: web1, etc.
72 // We can iterate through registers first to group them as in Figure 11
73 for (int r = 0; r < result.numColorsUsed; ++r) {
74 for (const auto &web : webs) {
75 auto it = result.webToRegister.find(web.id);
76 if (it != result.webToRegister.end() && it->second == r) {
77 outFile << "r" << r << ": web" << web.id << "\n";
78 }
79 }
80 }
81
82 // Output M: webY for any spilled webs (if any, although basic allocation
83 // might just fail)
84 for (const auto &web : webs) {
85 auto it = result.webToRegister.find(web.id);
86 if (it == result.webToRegister.end() || it->second == -1) {
87 outFile << "M: web" << web.id << "\n";
88 }
89 }
90 }
91
92 outFile.close();
93 return true;
94}
