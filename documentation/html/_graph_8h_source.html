34/**
35 * @brief Represents a vertex (node) in the graph.
36 *
37 * Stores the vertex info value, outgoing/incoming edge lists, and a set of
38 * auxiliary fields used by graph algorithms (BFS visited flag, DFS/Tarjan
39 * low-link numbers, Dijkstra distance, topological sort indegree, etc.).
40 *
41 * @tparam T Type of the info value stored in each vertex.
42 */
43template <class T>
44class Vertex {
45public:
46 Vertex(T in); ///< @brief Constructs a vertex with the given info value. @complexity O(1)
47 bool operator<(Vertex<T> & vertex) const; ///< @brief Comparison by `dist` field, required by MutablePriorityQueue.
48
49 T getInfo() const; ///< @brief Returns the vertex info. @complexity O(1)
50 std::vector<Edge<T> *> getAdj() const; ///< @brief Returns the list of outgoing edges. @complexity O(1)
51 bool isVisited() const; ///< @brief Returns the visited flag. @complexity O(1)
52 bool isProcessing() const; ///< @brief Returns the processing flag (DFS cycle detection). @complexity O(1)
53 unsigned int getIndegree() const; ///< @brief Returns the indegree (for topological sort). @complexity O(1)
54 double getDist() const; ///< @brief Returns the distance estimate (Dijkstra / Prim). @complexity O(1)
55 Edge<T> *getPath() const; ///< @brief Returns the predecessor edge on the shortest-path tree. @complexity O(1)
56 std::vector<Edge<T> *> getIncoming() const; ///< @brief Returns the list of incoming edges. @complexity O(1)
57
58 void setInfo(T info); ///< @brief Sets the vertex info. @complexity O(1)
59 void setVisited(bool visited); ///< @brief Sets the visited flag. @complexity O(1)
60 void setProcessing(bool processing); ///< @brief Sets the processing flag. @complexity O(1)
61
62 int getLow() const; ///< @brief Returns the Tarjan low-link value. @complexity O(1)
63 void setLow(int value); ///< @brief Sets the Tarjan low-link value. @complexity O(1)
64 int getNum() const; ///< @brief Returns the Tarjan discovery number. @complexity O(1)
65 void setNum(int value); ///< @brief Sets the Tarjan discovery number. @complexity O(1)
66
67 void setIndegree(unsigned int indegree); ///< @brief Sets the indegree counter. @complexity O(1)
68 void setDist(double dist); ///< @brief Sets the distance estimate. @complexity O(1)
69 void setPath(Edge<T> *path); ///< @brief Sets the predecessor edge. @complexity O(1)
70
71 /**
72 * @brief Adds an outgoing edge from this vertex to @p dest with weight @p w.
73 * @param dest Destination vertex pointer.
74 * @param w Edge weight.
75 * @return Pointer to the newly created Edge.
76 * @complexity O(1)
77 */
78 Edge<T> * addEdge(Vertex<T> *dest, double w);
79
80 /**
81 * @brief Removes the outgoing edge to the vertex with info @p in.
82 * @param in Info of the destination vertex to disconnect.
83 * @return `true` if at least one edge was removed, `false` if none found.
84 * @complexity O(deg(v)) where deg(v) is the out-degree of this vertex.
85 */
86 bool removeEdge(T in);
87
88 /**
89 * @brief Removes all outgoing edges from this vertex.
90 * @complexity O(deg(v))
91 */
92 void removeOutgoingEdges();
93
94 friend class MutablePriorityQueue<Vertex>;
95protected:
96 T info; ///< Info stored in the vertex (e.g., web ID).
97 std::vector<Edge<T> *> adj; ///< Outgoing edge list.
98
99 // auxiliary fields
100 bool visited = false; ///< Visited flag used by DFS/BFS.
101 bool processing = false; ///< Processing flag used by isDAG cycle detection.
102 int low = -1, num = -1; ///< Tarjan SCC low-link and discovery numbers.
103 unsigned int indegree; ///< Indegree counter used by topological sort.
104 double dist = 0; ///< Distance estimate used by Dijkstra/Prim.
105 Edge<T> *path = nullptr; ///< Predecessor edge on the shortest-path tree.
106 std::vector<Edge<T> *> incoming; ///< Incoming edge list.
107 int queueIndex = 0; ///< Index used by MutablePriorityQueue.
108
109 /**
110 * @brief Deletes an edge object and removes it from the destination's incoming list.
111 * @param edge Pointer to the edge to delete.
112 * @complexity O(deg_in(dest)) where deg_in is the in-degree of the destination.
113 */
114 void deleteEdge(Edge<T> *edge);
115};

155/**
156 * @brief Templated directed graph using an adjacency-list representation.
157 *
158 * Vertices are heap-allocated and stored in `vertexSet`. Edges are heap-
159 * allocated inside each Vertex's `adj` list. The graph owns all vertex and
160 * edge memory; the destructor frees it.
161 *
162 * In this project the graph is instantiated as `Graph<int>` where the int
163 * is the web ID (see GraphBuilder).
164 *
165 * @tparam T Type of the info value stored in each vertex.
166 */
167template <class T>
168class Graph {
169public:
170 ~Graph();
171
172 /**
173 * @brief Finds and returns the vertex whose info equals @p in.
174 * @param in Info value to search for.
175 * @return Pointer to the matching Vertex, or `nullptr` if not found.
176 * @complexity O(V) where V = number of vertices.
177 */
178 Vertex<T> *findVertex(const T &in) const;
179
180 /**
181 * @brief Adds a new vertex with info @p in to the graph.
182 * @param in Info value for the new vertex.
183 * @return `true` on success; `false` if a vertex with that info already exists.
184 * @complexity O(V) due to the duplicate check via findVertex().
185 */
186 bool addVertex(const T &in);
187
188 /**
189 * @brief Removes the vertex with info @p in and all its incident edges.
190 * @param in Info of the vertex to remove.
191 * @return `true` if the vertex was found and removed; `false` otherwise.
192 * @complexity O(V + E) — must remove all incoming edges from neighbours.
193 */
194 bool removeVertex(const T &in);
195
196 /**
197 * @brief Adds a directed edge from @p sourc to @p dest with weight @p w.
198 * @return `true` on success; `false` if either endpoint does not exist.
199 * @complexity O(V) for the two findVertex() calls.
200 */
201 bool addEdge(const T &sourc, const T &dest, double w);
202
203 /**
204 * @brief Removes the directed edge from @p source to @p dest.
205 * @return `true` if the edge existed and was removed; `false` otherwise.
206 * @complexity O(V + deg(source)).
207 */
208 bool removeEdge(const T &source, const T &dest);
209
210 /**
211 * @brief Adds a bidirectional (undirected) edge between @p sourc and @p dest.
212 *
213 * Creates two directed edges and links them via their `reverse` pointers.
214 *
215 * @return `true` on success; `false` if either endpoint does not exist.
216 * @complexity O(V) for the two findVertex() calls.
217 */
218 bool addBidirectionalEdge(const T &sourc, const T &dest, double w);
219
220 /**
221 * @brief Returns the number of vertices in the graph.
222 * @complexity O(1)
223 */
224 int getNumVertex() const;
225
226 /**
227 * @brief Returns the full vertex set.
228 * @return Reference to the internal vector of vertex pointers.
229 * @complexity O(1)
230 */
231 std::vector<Vertex<T> *> getVertexSet() const;
232
233
234protected:
235 std::vector<Vertex<T> *> vertexSet; ///< All vertices owned by this graph.
236
237 double ** distMatrix = nullptr; ///< Distance matrix for Floyd-Warshall.
238 int **pathMatrix = nullptr; ///< Path matrix for Floyd-Warshall.
239
240 /**
241 * @brief Returns the index of the vertex with info @p in in vertexSet.
242 * @return Index, or -1 if not found.
243 * @complexity O(V)
244 */
245 int findVertexIdx(const T &in) const;
246};

270 * Auxiliary function to add an outgoing edge to a vertex (this),

273template <class T>
274Edge<T> * Vertex<T>::addEdge(Vertex<T> *d, double w) {
275 auto newEdge = new Edge<T>(this, d, w);
276 adj.push_back(newEdge);
277 d->incoming.push_back(newEdge);
278 return newEdge;
279}
280
281/*
282 * Auxiliary function to remove an outgoing edge (with a given destination (d))
283 * from a vertex (this).
284 * Returns true if successful, and false if such edge does not exist.
285 */
286template <class T>
287bool Vertex<T>::removeEdge(T in) {
288 bool removedEdge = false;
289 auto it = adj.begin();
290 while (it != adj.end()) {
291 Edge<T> *edge = *it;
292 Vertex<T> *dest = edge->getDest();
293 if (dest->getInfo() == in) {
294 it = adj.erase(it);
295 deleteEdge(edge);
296 removedEdge = true; // allows for multiple edges to connect the same pair of vertices (multigraph)
297 }
298 else {
299 it++;
300 }
301 }
302 return removedEdge;
303}
304
305/*
306 * Auxiliary function to remove an outgoing edge of a vertex.
307 */
308template <class T>
309void Vertex<T>::removeOutgoingEdges() {
310 auto it = adj.begin();
311 while (it != adj.end()) {
312 Edge<T> *edge = *it;
313 it = adj.erase(it);
314 deleteEdge(edge);
315 }
316}

413template <class T>
414void Vertex<T>::deleteEdge(Edge<T> *edge) {
415 Vertex<T> *dest = edge->getDest();
416 // Remove the corresponding edge from the incoming list
417 auto it = dest->incoming.begin();
418 while (it != dest->incoming.end()) {
419 if ((*it)->getOrig()->getInfo() == info) {
420 it = dest->incoming.erase(it);
421 }
422 else {
423 it++;
424 }
425 }
426 delete edge;
427}

512/*
513 * Adds a vertex with a given content or info (in) to a graph (this).
514 * Returns true if successful, and false if a vertex with that content already exists.
515 */
516template <class T>
517bool Graph<T>::addVertex(const T &in) {

519 return false;

521 return true;

524/*
525 * Removes a vertex with a given content (in) from a graph (this), and
526 * all outgoing and incoming edges.
527 * Returns true if successful, and false if such vertex does not exist.
528 */
529template <class T>
530bool Graph<T>::removeVertex(const T &in) {
531 for (auto it = vertexSet.begin(); it != vertexSet.end(); it++) {
532 if ((*it)->getInfo() == in) {
533 auto v = *it;
534 v->removeOutgoingEdges();
535 for (auto u : vertexSet) {
536 u->removeEdge(v->getInfo());
537 }
538 vertexSet.erase(it);
539 delete v;
540 return true;
541 }
542 }
543 return false;
544}

546/*
547 * Adds an edge to a graph (this), given the contents of the source and
548 * destination vertices and the edge weight (w).
549 * Returns true if successful, and false if the source or destination vertex does not exist.
550 */
551template <class T>
552bool Graph<T>::addEdge(const T &sourc, const T &dest, double w) {
553 auto v1 = findVertex(sourc);
554 auto v2 = findVertex(dest);
555 if (v1 == nullptr || v2 == nullptr)
556 return false;
557 v1->addEdge(v2, w);
558 return true;
559}

561/*
562 * Removes an edge from a graph (this).
563 * The edge is identified by the source (sourc) and destination (dest) contents.
564 * Returns true if successful, and false if such edge does not exist.
565 */
566template <class T>
567bool Graph<T>::removeEdge(const T &sourc, const T &dest) {
568 Vertex<T> * srcVertex = findVertex(sourc);
569 if (srcVertex == nullptr) {
570 return false;
571 }

575template <class T>
576bool Graph<T>::addBidirectionalEdge(const T &sourc, const T &dest, double w) {
577 auto v1 = findVertex(sourc);
578 auto v2 = findVertex(dest);
579 if (v1 == nullptr || v2 == nullptr)
580 return false;
581 auto e1 = v1->addEdge(v2, w);
582 auto e2 = v2->addEdge(v1, w);
583 e1->setReverse(e2);
584 e2->setReverse(e1);
585 return true;
586}
